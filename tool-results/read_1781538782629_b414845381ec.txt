   155→function fillTemplate(template: string, order: Order): string {
   156→  const date = order.scheduledDate ? format(parseISO(order.scheduledDate), "dd MMM yyyy (EEE)") : "TBD";
   157→  return template
   158→    .replace(/\{customerName\}/g, order.customerName)
   159→    .replace(/\{date\}/g, date)
   160→    .replace(/\{address\}/g, order.address)
   161→    .replace(/\{phone\}/g, order.phone)
   162→    .replace(/\{orderId\}/g, order.orderId)
   163→    .replace(/\{size\}/g, order.size)
   164→    .replace(/\{points\}/g, order.points.toString())
   165→    .replace(/\{city\}/g, order.city)
   166→    .replace(/\{notes\}/g, order.notes || "N/A");
   167→}
   168→
   169→function formatPhoneForWhatsApp(phone: string, prefix: string = "60"): string {
   170→  if (phone.startsWith("+")) return phone.substring(1);
   171→  if (phone.startsWith(prefix)) return phone;
   172→  return `${prefix}${phone.replace(/^0/, "")}`;
   173→}
   174→
   175→function getWhatsAppLink(order: Order, template?: string, phonePrefix?: string): string {
   176→  const msg = template ? fillTemplate(template, order) : fillTemplate(DEFAULT_WHATSAPP_TEMPLATES[0].message, order);
   177→  const phone = formatPhoneForWhatsApp(order.phone, phonePrefix || "60");
   178→  return `https://wa.me/${phone}?text=${encodeURIComponent(msg)}`;
   179→}

   736→// ============ DATA FETCHING HOOK ============
   737→function useFetchData<T>(url: string) {
   738→  const [data, setData] = useState<T | null>(null);
   739→  const [error, setError] = useState<string | null>(null);
   740→  const [isLoading, setIsLoading] = useState(false);
   741→  const [version, setVersion] = useState(0);
   742→  const refetch = useCallback(() => setVersion(v => v + 1), []);
   743→  useEffect(() => {
   744→    if (!url) return;
   745→    let cancelled = false;
   746→    fetch(url)
   747→      .then(r => {
   748→        if (!cancelled) { setIsLoading(true); setError(null); }
   749→        if (!r.ok) throw new Error(`HTTP ${r.status}`);
   750→        return r.json();
   751→      })
   752→      .then(d => { if (!cancelled) { setData(d); setError(null); } })
   753→      .catch(e => { if (!cancelled) setError(e.message); })
   754→      .finally(() => { if (!cancelled) setIsLoading(false); });
   755→    return () => { cancelled = true; };
   756→  }, [url, version]);
   757→  return { data, error, isLoading, refetch };
   758→}

   771→// ============ ZONE BADGE ============
   772→function ZoneBadge({ zone, compact, userZones, isDisabled }: { zone: number; compact?: boolean; userZones?: UserZoneData[]; isDisabled?: boolean }) {
   773→  const z = ZONES[zone];
   774→  const userZone = userZones?.find(uz => uz.zoneId === zone);
   775→
   776→  // For custom zones (zoneId >= 100) with no ZONES entry and no userZone, return null
   777→  if (!z && !userZone) return null;
   778→
   779→  // Determine display name: user override first, then ZONES fallback
   780→  const name = userZone?.name || z?.name || `Zone ${zone}`;
   781→
   782→  // Determine colors: built-in zones use ZONES colors, custom zones use getZoneColor
   783→  const isCustomZone = zone >= CUSTOM_ZONE_START;
   784→  const customColors = isCustomZone ? getZoneColor(zone - CUSTOM_ZONE_START) : null;
   785→  const bgColor = customColors?.bgColor || z?.bgColor || "bg-slate-500/15";
   786→  const color = customColors?.color || z?.color || "text-slate-700";
   787→  const borderColor = customColors?.borderColor || z?.borderColor || "border-slate-500/30";
   788→
   789→  return (
   790→    <span className={`inline-flex items-center gap-1 rounded-md border px-1.5 py-0.5 text-[11px] font-medium ${bgColor} ${color} ${borderColor} ${isDisabled ? "opacity-50 line-through" : ""}`}>
   791→      <span className="w-1.5 h-1.5 rounded-full" style={{ backgroundColor: `oklch(0.7 0.14 ${zone * 40 + 100})` }} />
   792→      {compact ? name : `Z${zone} ${name}`}
   793→      {isDisabled && !compact && <span className="text-[9px] opacity-70 ml-0.5">(off)</span>}
   794→    </span>
   795→  );
   796→}

   809→// ============ MINI CALENDAR ============
   810→function MiniCalendar({
   811→  selectedDate,
   812→  onSelectDate,
   813→  holidays = [],
   814→  offDays = [],
   815→  isOffice = false,
   816→}: {
   817→  selectedDate: string;
   818→  onSelectDate: (date: string) => void;
   819→  holidays?: Holiday[];
   820→  offDays?: OffDay[];
   821→  isOffice?: boolean;
   822→}) {
   823→  const [viewMonth, setViewMonth] = useState(() => {
   824→    if (selectedDate) return parseISO(selectedDate);
   825→    return new Date();
   826→  });
   827→
   828→  const monthStart = startOfMonth(viewMonth);
   829→  const monthEnd = endOfMonth(viewMonth);
   830→  const calStart = startOfWeek(monthStart, { weekStartsOn: 1 });
   831→  const calEnd = endOfWeek(monthEnd, { weekStartsOn: 1 });
   832→  const days = eachDayOfInterval({ start: calStart, end: calEnd });
   833→
   834→  const holidaySet = useMemo(() => new Set(holidays.map(h => h.date)), [holidays]);
   835→  const offDaySet = useMemo(() => new Set(offDays.map(o => o.date)), [offDays]);
   836→  const todayStr = format(new Date(), "yyyy-MM-dd");
   837→
   838→  return (
   839→    <div>
   840→      {/* Month navigation */}
   841→      <div className="flex items-center justify-between mb-2">
   842→        <Button variant="ghost" size="sm" onClick={() => setViewMonth(addMonths(viewMonth, -1))} className="h-9 w-9 p-0 hover:bg-white/10">
   843→          <ChevronLeft className="h-4 w-4" />
   844→        </Button>
   845→        <span className="text-sm font-semibold text-foreground">{format(viewMonth, "MMMM yyyy")}</span>
   846→        <Button variant="ghost" size="sm" onClick={() => setViewMonth(addMonths(viewMonth, 1))} className="h-9 w-9 p-0 hover:bg-white/10">
   847→          <ChevronRight className="h-4 w-4" />
   848→        </Button>
   849→      </div>
   850→      {/* Day headers */}
   851→      <div className="grid grid-cols-7 gap-0.5 mb-0.5">
   852→        {["M", "T", "W", "T", "F", "S", "S"].map((d, i) => (
   853→          <div key={i} className="text-center text-[10px] font-medium text-muted-foreground py-1">{d}</div>
   854→        ))}
   855→      </div>
   856→      {/* Day cells */}
   857→      <div className="grid grid-cols-7 gap-0.5">
   858→        {days.map(day => {
   859→          const dateStr = format(day, "yyyy-MM-dd");
   860→          const isCurrentMonth = isSameMonth(day, viewMonth);
